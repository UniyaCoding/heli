local heli = script.Parent
local speed = 0.8 -- kecepatan maju
local bobSpeed = 2 -- kecepatan naik-turun
local bobHeight = 0.5 -- tinggi naik-turun
local radius = 80 -- ukuran lingkaran diperbesar (lebih luas)
local angularSpeed = 0.4 -- kecepatan rotasi melingkar (lebih lambat supaya halus)

heli.PrimaryPart.Anchored = true
task.wait(6) -- diam dulu 6 detik

while true do
	-- ====== Terbang lurus ======
	local flyDuration = math.random(5, 10)
	local flyStart = tick()
	while tick() - flyStart < flyDuration do
		local dir = heli.PrimaryPart.CFrame.LookVector
		local pos = heli.PrimaryPart.Position
			+ dir * speed
			+ Vector3.new(0, math.sin(tick() * bobSpeed) * bobHeight, 0)
		heli:MoveTo(pos)
		task.wait(0.03)
	end

	-- ====== Berputar melingkar luas seperti angka 0 ======
	local circleDuration = math.random(6, 10)
	local angle = 0
	local center = heli.PrimaryPart.Position

	while angle < math.pi * 2 * (circleDuration / 6) do
		angle += angularSpeed * 0.05
		local x = math.cos(angle) * radius
		local z = math.sin(angle) * radius
		local y = math.sin(angle * bobSpeed) * bobHeight

		local pos = center + Vector3.new(x, y, z)
		heli:MoveTo(pos)

		-- Hadapkan heli ke arah gerakan
		local nextPos = center + Vector3.new(math.cos(angle + 0.1) * radius, y, math.sin(angle + 0.1) * radius)
		local lookAt = CFrame.lookAt(pos, nextPos)
		heli:SetPrimaryPartCFrame(lookAt)

		task.wait(0.03)
	end
end
